import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: DefaultTabController(
        length: 3,
        child: Scaffold(
          appBar: AppBar(
            title: const Text('Tabs Demo'),
            bottom: const TabBar(
              tabs: [
                Tab(icon: Icon(Icons.directions_car), text: 'Car'),
                Tab(icon: Icon(Icons.directions_transit), text: 'Transit'),
                Tab(icon: Icon(Icons.directions_bike), text: 'Bike'),
              ],
            ),
          ),
          body: TabBarView(
            children: [
              tab1(),     // 👈 เรียกใช้ method ที่อยู่ข้างล่าง
              tab2(),
              tab3(),
            ],
          ),
        ),
      ),
    );
  }

  // ⬇️ Method แยกไว้ในไฟล์เดียวกัน (อยู่ใน class MyApp เหมือนกัน)
  Widget tab1() {
    return const Center(
      child: Text('This is Car Tab'),
    );
  }

  Widget tab2() {
    return const Center(
      child: Text('This is Transit Tab'),
    );
  }

  Widget tab3() {
    return const Center(
      child: Text('This is Bike Tab'),
    );
  }
}
